- [ ] **Step 4: 实现状态、全局表和格式化错误**

创建 `src/vm/ms_state.c`。错误格式化用两次 `vsnprintf`，并始终 `va_end` 每个副本：

```c
#include "vm/ms_state_internal.h"

#include <stdarg.h>
#include <stdio.h>
#include <string.h>

#include "core/ms_memory.h"

static void msStateRaiseV(MsState* L, MsResult result, const char* prefix,
    const char* fmt, va_list args) {
  va_list measure;
  va_copy(measure, args);
  int bodyLen = vsnprintf(NULL, 0, fmt, measure);
  va_end(measure);
  L->error = NULL;
  L->errorResult = result;
  if (bodyLen < 0) {
    msRaiseRuntimeError(L, "error formatting failed");
    return;
  }
  size_t prefixLen = strlen(prefix);
  char* text = msAlloc(prefixLen + (size_t)bodyLen + 1);
  if (text == NULL) {
    L->errorResult = MS_ERROR_OOM;
    return;
  }
  memcpy(text, prefix, prefixLen);
  vsnprintf(text + prefixLen, (size_t)bodyLen + 1, fmt, args);
  L->error = msStringNew(&L->heap, text, prefixLen + (size_t)bodyLen);
  msFree(text);
  if (L->error == NULL) {
    L->errorResult = MS_ERROR_OOM;
  }
}

void msRaiseRuntimeError(MsState* L, const char* fmt, ...) {
  va_list args;
  va_start(args, fmt);
  msStateRaiseV(L, MS_ERROR_RUNTIME, "", fmt, args);
  va_end(args);
}

void msRaiseTypeError(MsState* L, const char* fmt, ...) {
  va_list args;
  va_start(args, fmt);
  msStateRaiseV(L, MS_ERROR_RUNTIME, "TypeError: ", fmt, args);
  va_end(args);
}

MsState* msNewState(void) {
  MsState* L = msAlloc(sizeof(*L));
  if (L == NULL) {
    return NULL;
  }
  memset(L, 0, sizeof(*L));
  if (msHeapInit(&L->heap) != MS_OK) {
    msFree(L);
    return NULL;
  }
  return L;
}

void msCloseState(MsState* L) {
  if (L == NULL) {
    return;
  }
  msFree(L->globals);
  msHeapDestroy(&L->heap);
  msFree(L);
}

struct MsObject* msStateGetGlobal(MsState* L, const struct MsString* name) {
  for (int i = 0; i < L->globalsLen; ++i) {
    if (msStringEquals(L->globals[i].name, name)) {
      return L->globals[i].value;
    }
  }
  return NULL;
}

MsResult msStateSetGlobal(MsState* L, const struct MsString* name, struct MsObject* value) {
  for (int i = 0; i < L->globalsLen; ++i) {
    if (msStringEquals(L->globals[i].name, name)) {
      L->globals[i].value = value;
      return MS_OK;
    }
  }
  if (L->globalsLen == L->globalsCap) {
    int cap = L->globalsCap == 0 ? 8 : L->globalsCap * 2;
    void* entries = msRealloc(L->globals, (size_t)cap * sizeof(*L->globals));
    if (entries == NULL) {
      L->errorResult = MS_ERROR_OOM;
      return MS_ERROR_OOM;
    }
    L->globals = entries;
    L->globalsCap = cap;
  }
  L->globals[L->globalsLen].name = (struct MsString*)name;
  L->globals[L->globalsLen].value = value;
  ++L->globalsLen;
  return MS_OK;
}

const char* msStateErrorMessage(const MsState* L) {
  return L->error == NULL ? NULL : msStringData(L->error);
}

MsResult msStateErrorResult(const MsState* L) {
  return L->errorResult;
}

void msStateClearError(MsState* L) {
  L->error = NULL;
  L->errorResult = MS_OK;
}

struct MsObject* msNewNil(MsState* L) {
  return msHeapNil(&L->heap);
}

struct MsObject* msNewBool(MsState* L, bool value) {
  return msHeapBool(&L->heap, value);
}

struct MsObject* msNewInt(MsState* L, int64_t value) {
  struct MsObject* result = (struct MsObject*)msIntNew(&L->heap, value);
  if (result == NULL) {
    L->errorResult = MS_ERROR_OOM;
  }
  return result;
}

struct MsObject* msNewFloat(MsState* L, double value) {
  struct MsObject* result = (struct MsObject*)msFloatNew(&L->heap, value);
  if (result == NULL) {
    L->errorResult = MS_ERROR_OOM;
  }
  return result;
}
```

```c
struct MsObject* msNewFunction(MsState* L, struct MsProto* proto) {
```

```c

#include <math.h>
#include <string.h>

#include "core/ms_memory.h"
#include "object/ms_string.h"
#include "vm/ms_state_internal.h"

static struct MsObject* msOperatorError(MsState* L, const char* message) {
  msRaiseRuntimeError(L, "%s", message);
  return NULL;
}

static struct MsObject* msOperatorTypeError(MsState* L, MsOpCode op) {
  msRaiseTypeError(L, "unsupported operands for %s", msOpName(op));
  return NULL;
}

static bool msOperatorIsNumber(const struct MsObject* value) {
  MsTypeTag tag = msObjectTag(value);
  return tag == MS_TYPE_INT || tag == MS_TYPE_FLOAT;
}

static double msOperatorDouble(const struct MsObject* value) {
  return msObjectTag(value) == MS_TYPE_INT
      ? (double)msIntWordValue(value)
      : msFloatValue(value);
}

static struct MsObject* msOperatorConcat(MsState* L,
    const struct MsString* lhs, const struct MsString* rhs) {
  size_t aLen = msStringLen(lhs);
  size_t bLen = msStringLen(rhs);
  if (aLen + bLen == 0) {
    struct MsString* empty = msStringNew(&L->heap, NULL, 0);
    if (empty == NULL) {
      L->errorResult = MS_ERROR_OOM;
    }
    return (struct MsObject*)empty;
  }
  char* data = msAlloc(aLen + bLen);
  if (data == NULL) {
    L->errorResult = MS_ERROR_OOM;
    return NULL;
  }
  memcpy(data, msStringData(lhs), aLen);
  memcpy(data + aLen, msStringData(rhs), bLen);
  struct MsString* result = msStringNew(&L->heap, data, aLen + bLen);
  msFree(data);
  if (result == NULL) {
    L->errorResult = MS_ERROR_OOM;
    return NULL;
  }
  return (struct MsObject*)result;
}

static int msOperatorStringCompare(const struct MsString* a, const struct MsString* b) {
  size_t aLen = msStringLen(a);
  size_t bLen = msStringLen(b);
  size_t common = aLen < bLen ? aLen : bLen;
  int cmp = common == 0 ? 0 : memcmp(msStringData(a), msStringData(b), common);
  if (cmp != 0) {
    return cmp;
  }
  return aLen < bLen ? -1 : aLen > bLen ? 1 : 0;
}

static bool msOperatorCompareResult(MsOpCode op, int cmp) {
  switch (op) {
    case MS_OP_EQ: return cmp == 0;
    case MS_OP_NE: return cmp != 0;
    case MS_OP_LT: return cmp < 0;
    case MS_OP_LE: return cmp <= 0;
    case MS_OP_GT: return cmp > 0;
    case MS_OP_GE: return cmp >= 0;
    default: return false;
  }
}

struct MsObject* msObjectBinaryOp(MsState* L, MsOpCode op,
    struct MsObject* lhs, struct MsObject* rhs) {
  if (op == MS_OP_ADD && msObjectTag(lhs) == MS_TYPE_STRING
      && msObjectTag(rhs) == MS_TYPE_STRING) {
    return msOperatorConcat(L, (const struct MsString*)lhs, (const struct MsString*)rhs);
  }
  if (msObjectTag(lhs) == MS_TYPE_STRING && msObjectTag(rhs) == MS_TYPE_STRING
      && op >= MS_OP_EQ && op <= MS_OP_GE) {
    return msNewBool(L, msOperatorCompareResult(op, msOperatorStringCompare(
        (const struct MsString*)lhs, (const struct MsString*)rhs)));
  }
  if (op == MS_OP_EQ || op == MS_OP_NE) {
    bool equal = msObjectValueEquals(lhs, rhs);
    return msNewBool(L, op == MS_OP_EQ ? equal : !equal);
  }
  if (!msOperatorIsNumber(lhs) || !msOperatorIsNumber(rhs)) {
    return msOperatorTypeError(L, op);
  }
  bool bothInt = msObjectTag(lhs) == MS_TYPE_INT && msObjectTag(rhs) == MS_TYPE_INT;
  if (bothInt && (op == MS_OP_ADD || op == MS_OP_SUB || op == MS_OP_MUL
      || op == MS_OP_POW || op == MS_OP_FLOORDIV || op == MS_OP_MOD || op == MS_OP_SHL)) {
    return msOperatorError(L, "integer overflow");
  }
  double a = msOperatorDouble(lhs);
  double b = msOperatorDouble(rhs);
  switch (op) {
    case MS_OP_ADD: return msNewFloat(L, a + b);
    case MS_OP_SUB: return msNewFloat(L, a - b);
    case MS_OP_MUL: return msNewFloat(L, a * b);
    case MS_OP_DIV:
      if (b == 0.0) {
        return msOperatorError(L, "division by zero");
      }
      return msNewFloat(L, a / b);
    case MS_OP_FLOORDIV:
      if (b == 0.0) {
        return msOperatorError(L, "division by zero");
      }
      return msNewFloat(L, floor(a / b));
    case MS_OP_MOD: {
      if (b == 0.0) {
        return msOperatorError(L, "modulo by zero");
      }
      double value = fmod(a, b);
      if (value != 0.0 && ((value < 0.0) != (b < 0.0))) {
        value += b;
      }
      return msNewFloat(L, value);
    }
    case MS_OP_POW: return msNewFloat(L, pow(a, b));
    default:
      return msOperatorTypeError(L, op);
  }
}

struct MsObject* msObjectUnaryOp(MsState* L, MsOpCode op, struct MsObject* value) {
  if (op == MS_OP_NOT) {
    return msNewBool(L, !msObjectIsTruthy(value));
  }
  if (op == MS_OP_NEG && msObjectTag(value) == MS_TYPE_FLOAT) {
    return msNewFloat(L, -msFloatValue(value));
  }
  if (msObjectTag(value) == MS_TYPE_INT) {
    if (op == MS_OP_NEG || op == MS_OP_SHL) {
      return msOperatorError(L, "integer overflow");
    }
    if (op == MS_OP_INVERT) {
      return msNewInt(L, ~msIntWordValue(value));
    }
  }
  return msOperatorTypeError(L, op);
}
```

```c
static MsResult msVmEnsureStack(struct MsCoroutine* co, int needed) {
  if (needed < 0 || needed > MS_VM_STACK_MAX_CAP) {
    return MS_ERROR_RUNTIME;
  }
  if (needed <= co->stackCap) {
    return MS_OK;
  }
  int cap = co->stackCap == 0 ? MS_VM_STACK_INIT_CAP : co->stackCap;
  while (cap < needed && cap < MS_VM_STACK_MAX_CAP) {
    cap = cap > MS_VM_STACK_MAX_CAP / 2 ? MS_VM_STACK_MAX_CAP : cap * 2;
  }
  if (cap < needed) {
    return MS_ERROR_RUNTIME;
  }
  void* stack = msRealloc(co->stack, (size_t)cap * sizeof(*co->stack));
  if (stack == NULL) {
    return MS_ERROR_OOM;
  }
  co->stack = stack;
  co->stackCap = cap;
  return MS_OK;
}

static MsResult msVmEnsureFrames(struct MsCoroutine* co, int needed) {
  if (needed < 0 || needed > MS_VM_FRAMES_MAX) {
    return MS_ERROR_RUNTIME;
  }
  if (needed <= co->framesCap) {
    return MS_OK;
  }
  int cap = co->framesCap == 0 ? MS_VM_FRAMES_INIT_CAP : co->framesCap;
  while (cap < needed && cap < MS_VM_FRAMES_MAX) {
    cap = cap > MS_VM_FRAMES_MAX / 2 ? MS_VM_FRAMES_MAX : cap * 2;
  }
  void* frames = msRealloc(co->frames, (size_t)cap * sizeof(*co->frames));
  if (frames == NULL) {
    return MS_ERROR_OOM;
  }
  co->frames = frames;
  co->framesCap = cap;
  return MS_OK;
}

static MsResult msVmPushFrameAt(MsState* L, struct MsCoroutine* co,
    struct MsProto* proto, int argc, struct MsObject** argv, int base,
    bool argsInPlace, struct MsUpvalue** upvalues, int upvalueCount) {
  if (proto == NULL || argc != proto->paramCount || proto->hasVarArgs || proto->hasKwArgs
      || proto->localCount < proto->paramCount || proto->localCount < 0 || proto->stackSize < 0
      || base < 1) {
    msRaiseRuntimeError(L, "invalid call frame");
    return MS_ERROR_RUNTIME;
  }
  int64_t windowEnd = (int64_t)base + proto->localCount + proto->stackSize;
  if (windowEnd > MS_VM_STACK_MAX_CAP) {
    msRaiseRuntimeError(L, "stack overflow");
    return MS_ERROR_RUNTIME;
  }
  MsResult result = msVmEnsureFrames(co, co->frameCount + 1);
  if (result == MS_ERROR_RUNTIME) {
    msRaiseRuntimeError(L, "call stack overflow");
    return result;
  }
  if (result != MS_OK) {
    L->errorResult = result;
    return result;
  }
  result = msVmEnsureStack(co, (int)windowEnd);
  if (result == MS_ERROR_RUNTIME) {
    msRaiseRuntimeError(L, "stack overflow");
    return result;
  }
  if (result != MS_OK) {
    L->errorResult = result;
    return result;
  }
  if (!argsInPlace) {
    co->stack[base - 1] = msNewNil(L);
    for (int i = 0; i < argc; ++i) {
      co->stack[base + i] = argv[i];
    }
  }
  for (int i = argc; i < proto->localCount; ++i) {
    co->stack[base + i] = msNewNil(L);
  }
  co->stackTop = base + proto->localCount;
  struct MsCallFrame* frame = &co->frames[co->frameCount++];
  frame->proto = proto;
  frame->pc = 0;
  frame->baseIndex = base;
  frame->upvalues = upvalues;
  frame->upvalueCount = upvalueCount;
  return MS_OK;
}

MsResult msVmPushFrame(MsState* L, struct MsCoroutine* co, struct MsProto* proto,
    int argc, struct MsObject** argv) {
  if (argc != 0 && argv == NULL) {
    msRaiseRuntimeError(L, "invalid arguments");
    return MS_ERROR_RUNTIME;
  }
  return msVmPushFrameAt(L, co, proto, argc, argv, co->stackTop + 1, false, NULL, 0);
}
```

```c
static MsResult msVmRaiseAt(MsState* L, struct MsCallFrame* frame, const char* message) {
  const char* file = frame != NULL && frame->proto->sourceFile != NULL
      ? msStringData(frame->proto->sourceFile)
      : "<unknown>";
  uint32_t line = frame != NULL ? msProtoLineAt(frame->proto, frame->pc - 1) : 0;

static MsResult msVmPush(MsState* L, struct MsCoroutine* co,
    struct MsCallFrame* frame, struct MsObject* value) {
  int limit = frame->baseIndex + frame->proto->localCount + frame->proto->stackSize;
  if (co->stackTop >= limit) {
    return msVmRaiseAt(L, frame, "evaluation stack overflow");
  }
  co->stack[co->stackTop++] = value;
  return MS_OK;
}

static struct MsObject* msVmPop(struct MsCoroutine* co, const struct MsCallFrame* frame) {
  MS_ASSERT(co->stackTop > frame->baseIndex + frame->proto->localCount);
  return co->stack[--co->stackTop];
```

```c
MsResult msVmRun(MsState* L, struct MsCoroutine* co, int stopAt) {
  int entryTop = co->stackTop;
  int entryFrames = co->frameCount;
  if (stopAt < 0 || stopAt > entryFrames) {
    msRaiseRuntimeError(L, "invalid VM stop depth");
    return MS_ERROR_RUNTIME;
  }
  if (stopAt == entryFrames) {
    return MS_OK;
  }
  struct MsCallFrame* frame = &co->frames[co->frameCount - 1];
  while (co->frameCount > stopAt) {
    if (frame->pc < 0 || frame->pc >= frame->proto->codeLen) {
      return msVmFinishError(L, co, stopAt, entryTop, msVmRaiseAt(L, frame, "invalid program counter"));
    }
    uint32_t instr = frame->proto->code[frame->pc++];
    MsResult result = MS_OK;
    switch (msOpDecodeOp(instr)) {
      case MS_OP_NOP:
        break;
      case MS_OP_LOAD_NIL:
        result = msVmPush(L, co, frame, msNewNil(L));
        break;
      case MS_OP_LOAD_TRUE:
        result = msVmPush(L, co, frame, msNewBool(L, true));
        break;
      case MS_OP_LOAD_FALSE:
        result = msVmPush(L, co, frame, msNewBool(L, false));
        break;
      case MS_OP_RETURN: {
        struct MsObject* value = msVmPop(co, frame);
        int returnBase = frame->baseIndex - 1;
        --co->frameCount;
        co->stackTop = returnBase;
        co->stack[co->stackTop++] = value;
        if (co->frameCount == stopAt) {
          return MS_OK;
        }
        frame = &co->frames[co->frameCount - 1];
        break;
      }
      default:
        MS_ASSERT(!"opcode not implemented");
        result = msVmRaiseAt(L, frame, "opcode not implemented");
        break;
    }
    if (result != MS_OK) {
      return msVmFinishError(L, co, stopAt, entryTop, result);
    }
  }
  return MS_OK;
}

MsResult msCallProto(MsState* L, struct MsCoroutine* co, struct MsProto* proto,
    int argc, struct MsObject** argv, struct MsObject** out) {
  if (out == NULL) {
    return MS_ERROR_RUNTIME;
  }
  *out = NULL;
  int stopAt = co->frameCount;
  int entryTop = co->stackTop;
  MsResult result = msVmPushFrame(L, co, proto, argc, argv);
  if (result != MS_OK) {
    co->stackTop = entryTop;
    return result;
  }
  result = msVmRun(L, co, stopAt);
  if (result == MS_OK) {
    MS_ASSERT(co->stackTop == entryTop + 1);
    *out = co->stack[entryTop];
  } else {
    co->stackTop = entryTop;
  }
  return result;
}
```

```c
static MsResult msVmBinaryOp(MsState* L, struct MsCoroutine* co,
    struct MsCallFrame* frame, MsOpCode op) {
  struct MsObject* rhs = msVmPop(co, frame);
  struct MsObject* lhs = msVmPop(co, frame);
  struct MsObject* value = NULL;
  if (msObjectTag(lhs) == MS_TYPE_INT && msObjectTag(rhs) == MS_TYPE_INT) {
    int64_t a = msIntWordValue(lhs);
    int64_t b = msIntWordValue(rhs);
    if (op == MS_OP_LT) {
      value = msNewBool(L, a < b);
    } else if (op == MS_OP_ADD
        && !((b > 0 && a > INT64_MAX - b) || (b < 0 && a < INT64_MIN - b))) {
      value = msNewInt(L, a + b);
    }
  }
  if (value == NULL && msStateErrorResult(L) == MS_OK) {
    value = msObjectBinaryOp(L, op, lhs, rhs);
  }
  if (value == NULL) {
    MsResult error = msStateErrorResult(L);
    return error == MS_ERROR_OOM ? error : MS_ERROR_RUNTIME;
  }
  return msVmPush(L, co, frame, value);
}
```

```c
static bool msVmAddOverflows(int64_t a, int64_t b) {
  return (b > 0 && a > INT64_MAX - b) || (b < 0 && a < INT64_MIN - b);
}

static bool msVmSubOverflows(int64_t a, int64_t b) {
  return (b > 0 && a < INT64_MIN + b) || (b < 0 && a > INT64_MAX + b);
}

static bool msVmMulOverflows(int64_t a, int64_t b) {
  if (a > 0) {
    return b > 0 ? a > INT64_MAX / b : b < INT64_MIN / a;
  }
  if (b > 0) {
    return a < INT64_MIN / b;
  }
  return a != 0 && b < INT64_MAX / a;
}

static bool msVmIntPow(int64_t a, int64_t b, int64_t* out) {
  int64_t result = 1;
  int64_t base = a;
  int64_t exponent = b;
  while (exponent > 0) {
    if ((exponent & 1) != 0) {
      if (msVmMulOverflows(result, base)) {
        return false;
      }
      result *= base;
    }
    exponent >>= 1;
    if (exponent != 0) {
      if (msVmMulOverflows(base, base)) {
        return false;
      }
      base *= base;
    }
  }
  *out = result;
  return true;
}

static bool msVmShiftLeft(int64_t a, int64_t b, int64_t* out) {
  if (b == 63) {
    if (a == 0 || a == -1) {
      *out = a == 0 ? 0 : INT64_MIN;
      return true;
    }
    return false;
  }
  int64_t factor = (int64_t)(1ULL << (unsigned)b);
  if (msVmMulOverflows(a, factor)) {
    return false;
  }
  *out = a * factor;
  return true;
}

static int64_t msVmShiftRight(int64_t a, int64_t b) {
  if (b == 63) {
    return a < 0 ? -1 : 0;
  }
  uint64_t bits = (uint64_t)a;
  if (a >= 0) {
    return (int64_t)(bits >> (unsigned)b);
  }
  return -1 - (int64_t)((~bits) >> (unsigned)b);
}
```

`msVmShiftLeft` 仅在 `0 <= b < 64` 时调用；`msVmMulOverflows` 的各除法分支均避开
`INT64_MIN / -1`。

```c
static MsResult msVmBinaryOp(MsState* L, struct MsCoroutine* co,
    struct MsCallFrame* frame, MsOpCode op) {
  struct MsObject* rhs = msVmPop(co, frame);
  struct MsObject* lhs = msVmPop(co, frame);
  MsTypeTag leftTag = msObjectTag(lhs);
  MsTypeTag rightTag = msObjectTag(rhs);
  bool leftNumber = leftTag == MS_TYPE_INT || leftTag == MS_TYPE_FLOAT;
  bool rightNumber = rightTag == MS_TYPE_INT || rightTag == MS_TYPE_FLOAT;
  struct MsObject* value = NULL;
  if (leftTag == MS_TYPE_INT && rightTag == MS_TYPE_INT) {
    int64_t a = msIntWordValue(lhs);
    int64_t b = msIntWordValue(rhs);
    int64_t integer = 0;
    bool haveInteger = false;
    switch (op) {
      case MS_OP_ADD:
        haveInteger = !msVmAddOverflows(a, b);
        if (haveInteger) integer = a + b;
        break;
      case MS_OP_SUB:
        haveInteger = !msVmSubOverflows(a, b);
        if (haveInteger) integer = a - b;
        break;
      case MS_OP_MUL:
        haveInteger = !msVmMulOverflows(a, b);
        if (haveInteger) integer = a * b;
        break;
      case MS_OP_FLOORDIV:
      case MS_OP_MOD:
        if (b == 0) {
          return msVmRaiseAt(L, frame, op == MS_OP_MOD ? "modulo by zero" : "division by zero");
        }
        if (a == INT64_MIN && b == -1) {
          break;
        }
        {
          int64_t quotient = a / b;
          int64_t remainder = a % b;
          if (remainder != 0 && ((remainder < 0) != (b < 0))) {
            --quotient;
            remainder += b;
          }
          integer = op == MS_OP_FLOORDIV ? quotient : remainder;
          haveInteger = true;
        }
        break;
      case MS_OP_POW:
        if (b < 0) {
          value = msNewFloat(L, pow((double)a, (double)b));
        } else {
          haveInteger = msVmIntPow(a, b, &integer);
        }
        break;
      case MS_OP_BITAND: integer = a & b; haveInteger = true; break;
      case MS_OP_BITOR: integer = a | b; haveInteger = true; break;
      case MS_OP_BITXOR: integer = a ^ b; haveInteger = true; break;
      case MS_OP_SHL:
      case MS_OP_SHR:
        if (b < 0 || b >= 64) {
          return msVmRaiseAt(L, frame, "shift count out of range");
        }
        if (op == MS_OP_SHR) {
          integer = msVmShiftRight(a, b);
          haveInteger = true;
        } else {
          haveInteger = msVmShiftLeft(a, b, &integer);
        }
        break;
      default:
        break;
    }
    if (haveInteger) {
      value = msNewInt(L, integer);
    }
  } else if (leftNumber && rightNumber) {
    double a = leftTag == MS_TYPE_INT ? (double)msIntWordValue(lhs) : msFloatValue(lhs);
    double b = rightTag == MS_TYPE_INT ? (double)msIntWordValue(rhs) : msFloatValue(rhs);
    switch (op) {
      case MS_OP_ADD: value = msNewFloat(L, a + b); break;
      case MS_OP_SUB: value = msNewFloat(L, a - b); break;
      case MS_OP_MUL: value = msNewFloat(L, a * b); break;
      case MS_OP_DIV:
        if (b == 0.0) return msVmRaiseAt(L, frame, "division by zero");
        value = msNewFloat(L, a / b);
        break;
      case MS_OP_POW: value = msNewFloat(L, pow(a, b)); break;
      default:
        break;
    }
  }
  if (op == MS_OP_DIV && leftTag == MS_TYPE_INT && rightTag == MS_TYPE_INT) {
    int64_t divisor = msIntWordValue(rhs);
    if (divisor == 0) {
      return msVmRaiseAt(L, frame, "division by zero");
    }
    value = msNewFloat(L, (double)msIntWordValue(lhs) / (double)divisor);
  }
  if (value == NULL && msStateErrorResult(L) == MS_OK) {
    value = msObjectBinaryOp(L, op, lhs, rhs);
  }
  if (value == NULL) {
    MsResult error = msStateErrorResult(L);
    return error == MS_ERROR_OOM ? error : MS_ERROR_RUNTIME;
  }
  return msVmPush(L, co, frame, value);
}
```

把 13 个二元算术 case 分组调用 `msVmBinaryOp`，每个 case 以 `// fallthrough` 标注直到公共
尾部。包含 `<math.h>`。普通 int `ADD/SUB/MUL` 在 `msObjectBinaryOp` 中也使用相同的溢出检查
或直接由 VM 快路径处理；不得把慢路径溢出改成 float。

```c
MS_TEST(vm, allocationFailuresRestoreAndDoNotLeak) {
  bool sawSuccess = false;
  for (int64_t failAfter = 0; failAfter < 40 && !sawSuccess; ++failAfter) {
    MsVmFixture fix;
    msVmFixtureInit(&fix);
    fix.b.proto.localCount = 1;
    int forty = msVmTestConst(&fix, msVmTestInt(&fix, 40));
    int two = msVmTestConst(&fix, msVmTestInt(&fix, 2));
    msProtoEmitABC(&fix.b, MS_OP_LOAD_CONST, 0, (uint16_t)forty);
    msProtoEmitABC(&fix.b, MS_OP_LOAD_CONST, 0, (uint16_t)two);
    msProtoEmitABC(&fix.b, MS_OP_ADD, 0, 0);
    msProtoEmitABC(&fix.b, MS_OP_RETURN, 0, 0);
    fix.status = msProtoBuild(&fix.b, &fix.proto);
    MS_ASSERT_EQ(MS_OK, fix.status);
    msMemSetFailAfter(failAfter);
    fix.status = msCallProto(fix.L, &fix.co, fix.proto, 0, NULL, &fix.result);
    msMemSetFailAfter(-1);
    if (fix.status == MS_OK) {
      sawSuccess = true;
      MS_ASSERT_EQ(42, msIntWordValue(fix.result));
    } else {
      MS_ASSERT_EQ(MS_ERROR_OOM, fix.status);
      MS_ASSERT_TRUE(fix.result == NULL);
      MS_ASSERT_EQ(0, fix.co.frameCount);
      MS_ASSERT_EQ(0, fix.co.stackTop);
    }
    msVmFixtureDestroy(&fix);
  }
  MS_ASSERT_TRUE(sawSuccess);
}
```
